   18:       ROLES_KEY,

   20:     );
   21:

   23:       return true;
   24:     }

   27:       user?: JwtPayload;
   28:     }>();
   29:

   31: 
   32:     if (!user || !requiredRoles.includes(user.role)) {
   33:       throw new ForbiddenException(
   34:         'You do not have permission to perform this action',

   36:     }
   37: 
   38:     return true;

    1: import { Injectable, UnauthorizedException } from '@nestjs/common';
    2: import { JwtService } from '@nestjs/jwt';
    3: import * as bcrypt from 'bcrypt';
    4: 
    5: import { AcademyMembership } from '../memberships/entities/academy-membership.entity';
    6: import { User, UserStatus } from '../users/entities/user.entity';
    7: import { UserProfile, UsersService } from '../users/users.service';
    8: import { LoginDto } from './dto/login.dto';
    9: import { JwtPayload } from './interfaces/jwt-payload.interface';
   10: 
   11: @Injectable()
   12: export class AuthService {
   13:   constructor(
   14:     private readonly usersService: UsersService,
   15:     private readonly jwtService: JwtService,
   16:   ) {}
   17: 
   18:   async login(loginDto: LoginDto) {
   19:     const user = await this.usersService.findByEmailWithPassword(
   20:       loginDto.email,
   21:     );
   22: 
   23:     if (!user) {
   24:       throw new UnauthorizedException('Invalid email or password');
   25:     }
   26: 
   27:     if (user.status !== UserStatus.ACTIVE) {
   28:       throw new UnauthorizedException('User account is not active');
   29:     }
   30: 
   31:     if (!user.passwordHash) {
   32:       throw new UnauthorizedException('Invalid email or password');
   33:     }
   34: 
   35:     const passwordMatches = await bcrypt.compare(
   36:       loginDto.password,
   37:       user.passwordHash,
   38:     );
   39: 
   40:     if (!passwordMatches) {
   41:       throw new UnauthorizedException('Invalid email or password');
   42:     }
   43: 
   44:     const membership = this.getPrimaryMembership(user);
   45: 
   46:     const payload: JwtPayload = {
   47:       sub: user.id,
   48:       email: user.email,
   49:       academyId: membership.academyId,
   50:       branchId: membership.branchId,
   51:       role: membership.role,
   52:     };
   53: 
   54:     const accessToken = await this.jwtService.signAsync(payload);
   55: 
   56:     await this.usersService.updateLastLogin(user.id);
   57: 
   58:     const profile: UserProfile = this.usersService.toProfile(user);
   59: 
   60:     return {
   61:       accessToken,
   62:       tokenType: 'Bearer',
   63:       expiresIn: 900,
   64:       user: profile,
   65:       activeMembership: {
   66:         id: membership.id,
   67:         academyId: membership.academyId,
   68:         branchId: membership.branchId,
   69:         role: membership.role,
   70:       },
   71:     };
   72:   }
   73: 
   74:   async me(userId: string): Promise<UserProfile> {
   75:     return this.usersService.findProfileById(userId);
   76:   }
   77: 
   78:   private getPrimaryMembership(user: User): AcademyMembership {
   79:     const activeMemberships = (user.memberships ?? []).filter(
   80:       (membership) => membership.isActive,
   81:     );
   82: 
   83:     const primaryMembership =
   84:       activeMemberships.find((membership) => membership.isPrimary) ??
   85:       activeMemberships[0];
   86: 
   87:     if (!primaryMembership) {
   88:       throw new UnauthorizedException('User has no active academy membership');
   89:     }
   90: 
   91:     return primaryMembership;
   92:   }
   93: }
